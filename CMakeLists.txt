cmake_minimum_required(VERSION 3.20)
project(Liberty VERSION 0.1.3 LANGUAGES CXX)

add_executable(Liberty WIN32 src/main.cpp app.rc)
target_compile_features(Liberty PRIVATE cxx_std_17)
target_compile_definitions(Liberty PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(Liberty PRIVATE
  user32 shell32 advapi32 gdi32 comctl32 comdlg32
  dwmapi gdiplus ole32 oleaut32 uxtheme wtsapi32
)

if(MSVC)
  # Static C/C++ runtime: the release remains one portable EXE.
  set_property(TARGET Liberty PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(Liberty PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(Liberty PRIVATE $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/Ob2>)
  target_link_options(Liberty PRIVATE /MANIFEST:NO /SUBSYSTEM:WINDOWS)
  target_link_options(Liberty PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
else()
  target_compile_options(Liberty PRIVATE -Wall -Wextra -Os -ffunction-sections -fdata-sections)
  target_link_options(Liberty PRIVATE -static -s -Wl,--gc-sections -mwindows)
endif()

install(TARGETS Liberty RUNTIME DESTINATION .)

include(CTest)
if(BUILD_TESTING AND MSVC)
  add_executable(LibertyRegression tests/regression.cpp app.rc)
  target_compile_features(LibertyRegression PRIVATE cxx_std_17)
  target_compile_definitions(LibertyRegression PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(LibertyRegression PRIVATE /W4 /permissive- /utf-8)
  target_link_options(LibertyRegression PRIVATE /MANIFEST:NO)
  target_link_libraries(LibertyRegression PRIVATE user32 shell32 advapi32 gdi32 comctl32 comdlg32 dwmapi gdiplus ole32 oleaut32 uxtheme wtsapi32)
  set_property(TARGET LibertyRegression PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME LibertyRegression COMMAND LibertyRegression)
  add_executable(LibertyDebugSession tools/debug-session.cpp)
  target_compile_features(LibertyDebugSession PRIVATE cxx_std_17)
  target_compile_options(LibertyDebugSession PRIVATE /W4 /utf-8)
  target_link_libraries(LibertyDebugSession PRIVATE dbghelp)
  add_executable(LibertyShutdownIntegration tests/shutdown-integration.cpp)
  target_compile_features(LibertyShutdownIntegration PRIVATE cxx_std_17)
  target_compile_definitions(LibertyShutdownIntegration PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_compile_options(LibertyShutdownIntegration PRIVATE /W4 /utf-8)
  target_link_libraries(LibertyShutdownIntegration PRIVATE advapi32)
endif()
